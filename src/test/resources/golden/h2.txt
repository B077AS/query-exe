    PRIMARY KEY ("id")

### create table with fk
CREATE TABLE "shop"."orders" (
    "id" INT GENERATED BY DEFAULT AS IDENTITY NOT NULL,
    "email" VARCHAR(120) NOT NULL UNIQUE,
    "price" DECIMAL(10,2) DEFAULT 0,
    "order" VARCHAR(20) DEFAULT 'new',
    "owner_id" INT,
    PRIMARY KEY ("id"),
    CONSTRAINT "fk_orders_owner" FOREIGN KEY ("owner_id") REFERENCES "shop"."owners" ("id") ON DELETE CASCADE ON UPDATE NO ACTION
);

### create table
CREATE TABLE `shop`.`orders` (
    `id` INT AUTO_INCREMENT NOT NULL,
    `email` VARCHAR(120) NOT NULL UNIQUE,
    `price` DECIMAL(10,2),
    `order` VARCHAR(20),
    PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

### create table with fk
CREATE TABLE `shop`.`orders` (
    `id` INT AUTO_INCREMENT NOT NULL,
    `email` VARCHAR(120) NOT NULL UNIQUE,
    `price` DECIMAL(10,2),
    `order` VARCHAR(20),
    `owner_id` INT,
    PRIMARY KEY (`id`),
    CONSTRAINT `fk_orders_owner` FOREIGN KEY (`owner_id`) REFERENCES `owners` (`id`) ON DELETE CASCADE ON UPDATE NO ACTION
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
